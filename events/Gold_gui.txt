add_namespace = USA_gold_for_fra

country_event = {
	id = USA_gold_for_fra.1
	title = USA_gold_for_fra.1t
	desc = USA_gold_for_fra.1d
	picture = GFX_report_event_france_tank_production

	is_triggered_only = yes

	option = {
		name = USA_gold_for_fra.a
		trigger = {
			USA = {
				NOT = {
					has_country_flag = FRA_no_gold
				}
			}
		}
		hidden_effect = {
			USA = {
				set_country_flag = FRA_no_gold
				country_event = USA_gold_for_fra.4
			}
		}
		custom_effect_tooltip = FRA_gold_brake_tt
	}
	
	option = {
		name = USA_gold_for_fra.b
		USA = {
			add_dynamic_modifier = {
				modifier = FRA_STEAL_GOLD_dm
			}
		}
		custom_effect_tooltip = generic_skip_one_line_tt
		FRA = {
			add_dynamic_modifier = {
				modifier = FRA_get_GOLD_dm
			}
		}
	}
}
